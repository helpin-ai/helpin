Embed Snippet

Copy this code and paste it before the closing </body> tag on your website.

<!-- Helpin Chat Widget -->
<script src="https://cdn.helpin.ai/widget/helpin-widget.min.js"></script>
<script>
  HelpinWidget.init({
    widgetKey: "your-widget-key",
    apiUrl: "https://helpin.yourapp.com",
    companyName: "Your Company",
    primaryColor: "#6C5CE7",
    theme: "light",
    greeting: "Hi! How can we help?",
    preChatForm: true
  });
</script>
